# P34 Onboarding Signal Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a first-login signal (`hasOnboarded`) to the `User` model, expose a `PATCH /api/me/onboarding` endpoint that sets only the caller's own flag, and thread the flag through the AppShell server component as a prop for future tour-engine consumption.

**Architecture:** A single additive `ALTER TABLE "User" ADD COLUMN "hasOnboarded" BOOLEAN NOT NULL DEFAULT false` migration keeps the DB change irreversible-safe (existing rows default to `false`). The endpoint reads `ctx.userId` exclusively from `withAuthScoped()` — no userId in the request body — making it impossible to flip another user's flag. AppShell already performs a `Promise.all` for org + membership data; the `hasOnboarded` query joins that batch. No login/signup/JWT/auth-boundary code is touched.

**Tech Stack:** Prisma 7 + PostgreSQL · Next.js 14 App Router · `withAuthScoped` RBAC layer · Upstash rate limiting (in-memory fallback) · Vitest

- [ ] **Step 2.1: Create the rate limiter**

Create `src/lib/ratelimit-onboarding.ts`:

```typescript
// 20 PATCH /api/me/onboarding requests per hour per user.
// Upstash sliding-window when configured; in-memory map fallback otherwise.
// Bypasses in NODE_ENV=test so unit tests do not need to manage state.

import { Ratelimit } from "@upstash/ratelimit"
import { Redis } from "@upstash/redis"

let _rl: Ratelimit | null = null

function getUpstashLimiter(): Ratelimit | null {
  if (!process.env.UPSTASH_REDIS_REST_URL || !process.env.UPSTASH_REDIS_REST_TOKEN) return null
  if (!_rl) {
    const redis = Redis.fromEnv()
    _rl = new Ratelimit({ redis, limiter: Ratelimit.slidingWindow(20, "3600 s"), prefix: "pulse:onboarding" })
  }
  return _rl
}

const HOUR_WINDOW_MS = 3_600_000
const _store = new Map<string, number[]>()

function memCheck(key: string): boolean {
  const now = Date.now()
  const windowStart = now - HOUR_WINDOW_MS
  const prev = _store.get(key) ?? []
  const inWindow = prev.filter((t) => t > windowStart)
  if (inWindow.length >= 20) {
    _store.set(key, inWindow)
    return false
  }
  inWindow.push(now)
  _store.set(key, inWindow)
  return true
}

export function _checkOnboardingMemRateLimit(userId: string): { allowed: boolean } {
  return { allowed: memCheck(userId) }
}

export async function checkOnboardingRateLimit(userId: string): Promise<{ allowed: boolean }> {
  if (process.env.NODE_ENV === "test") return { allowed: true }
  const upstash = getUpstashLimiter()
  if (upstash) {
    try {
      const result = await upstash.limit(userId)
      return { allowed: result.success }
    } catch {
      return { allowed: true }
    }
  }
  return _checkOnboardingMemRateLimit(userId)
}

export function _resetOnboardingRateLimitStore(): void {
  _store.clear()
}
```

import { NextResponse } from "next/server"
import { db } from "@/lib/db"
import { withAuthScoped } from "@/lib/withAuthScoped"
import { checkOnboardingRateLimit } from "@/lib/ratelimit-onboarding"

// Own-row-only guard: userId comes exclusively from the verified auth context.
// The request body carries only `hasOnboarded: boolean` — there is no userId
// field, so there is no path for one user to flip another user's flag.
export async function PATCH(request: Request) {
  const ctx = await withAuthScoped()
  if (!ctx) return NextResponse.json({ error: "Unauthorized" }, { status: 401 })

  const { allowed } = await checkOnboardingRateLimit(ctx.userId)
  if (!allowed) return NextResponse.json({ reason: "rate_limit_exceeded" }, { status: 429 })

  let body: unknown
  try {
    body = await request.json()
  } catch {
    return NextResponse.json({ error: "Invalid JSON" }, { status: 400 })
  }

  const raw = body as Record<string, unknown>
  if (typeof raw["hasOnboarded"] !== "boolean") {
    return NextResponse.json({ error: "hasOnboarded must be a boolean" }, { status: 400 })
  }

  await db.user.update({
    where: { id: ctx.userId },
    data: { hasOnboarded: raw["hasOnboarded"] },
  })

  return NextResponse.json({ ok: true })
}
```

### Why AppShell is the right read point

AppShell is the outermost server component wrapping every authenticated page. It already holds the auth context from `withAuthScoped()` and performs a `Promise.all` for org data. Reading `hasOnboarded` here means:
- It's fetched once per navigation, server-side, with zero client round-trips.
- `AppShellClient` (the tour engine's future host) receives it as a prop on mount.
- No coupling to JWT, login, or signup — the flag is read fresh from the DB on each dashboard load.

export async function AppShell({ role, userName, userId, children }: AppShellProps) {
  const ctx = await withAuthScoped()

  let org: { name: string } | null = null
  let membershipOptions: MembershipOption[] = []
  let pendingInviteCount = 0
  let hasOnboarded = false

  if (ctx) {
    const [orgRow, memberships, pendingCount, userRow] = await Promise.all([
      db.organisation.findUnique({ where: { id: ctx.organisationId }, select: { name: true } }),
      db.membership.findMany({
        where: { userId: ctx.userId, status: "ACTIVE" },
        select: {
          organisationId: true,
          role: true,
          teamId: true,
          organisation: { select: { name: true } },
        },
        orderBy: { createdAt: "asc" },
      }),
      db.membership.count({ where: { userId: ctx.userId, status: "PENDING" } }),
      db.user.findUnique({ where: { id: ctx.userId }, select: { hasOnboarded: true } }),
    ])
    org = orgRow
    membershipOptions = memberships.map((m) => ({
      organisationId: m.organisationId,
      organisationName: m.organisation?.name ?? "",
      role: m.role as MembershipOption["role"],
      teamId: m.teamId,
    }))
    pendingInviteCount = pendingCount
    hasOnboarded = userRow?.hasOnboarded ?? false
  }

  void userId

  return (
    <AppShellClient
      role={role}
      userName={userName}
      orgName={org?.name ?? ""}
      activeOrganisationId={ctx?.organisationId ?? ""}
      memberships={membershipOptions}
      pendingInviteCount={pendingInviteCount}
      hasOnboarded={hasOnboarded}
    >
      {children}
    </AppShellClient>
  )
}
```
