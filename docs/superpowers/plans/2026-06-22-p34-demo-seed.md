**Goal:** Create `prisma/seed-demo.ts` — a local-only, idempotent demo seed that populates `ActivityEvent`, `Intelligence`, `IntelligenceHighlight`, `MemberDailyTime`, `ReadinessSnapshot`, `CodeHealthSnapshot`, `ExecutiveSummary`, and `ContextDriftAssessment` rows so the 5 walkthrough screens render with convincing, honest-looking data.

**Architecture:** A standalone TypeScript script (same pattern as `seed.ts` / `seed1x.ts`) with two hard guards: `LOCAL_DEMO=1` env var + local DB hostname check. Reads org/team/project rows the base `seed.ts` already created, adds 5 demo users, then inserts all activity data in one idempotent pass detected via a `gitCommitSha: "demo-seed-v1"` marker on the first event. No new Prisma models, no migration, no change to existing seeds.

**Tech Stack:** TypeScript, Prisma 7 (`pg` adapter), `pg.Pool`, `bcryptjs`, `node:crypto`

// prisma/seed-demo.ts
// LOCAL-ONLY demo seed for the landing page walkthrough recording.
// NEVER run against a production database.
// Usage: LOCAL_DEMO=1 node --env-file .env --import tsx/esm prisma/seed-demo.ts

import { Pool } from "pg"
import { PrismaPg } from "@prisma/adapter-pg"
import { PrismaClient } from "../src/generated/prisma"
import bcrypt from "bcryptjs"
import { createHash } from "node:crypto"

// ── Safety guard — must execute before any DB connection ─────────────────────
function enforceLocalOnly(): void {
  if (process.env.LOCAL_DEMO !== "1") {
    console.error(
      "seed-demo.ts: LOCAL_DEMO=1 is required.\n" +
        "This script must never run against a production database.\n" +
        "Usage: LOCAL_DEMO=1 node --env-file .env --import tsx/esm prisma/seed-demo.ts"
    )
    process.exit(1)
  }

  const raw = process.env.DATABASE_URL ?? ""
  if (!raw) {
    console.error("seed-demo.ts: DATABASE_URL is not set.")
    process.exit(1)
  }

  let host: string
  try {
    host = new URL(raw).hostname
  } catch {
    console.error("seed-demo.ts: DATABASE_URL is not a valid URL.")
    process.exit(1)
  }

  const LOCAL_HOSTS = ["localhost", "127.0.0.1", "host.docker.internal", "postgres", "db"]
  if (!LOCAL_HOSTS.some((h) => host === h)) {
    console.error(
      `seed-demo.ts: refusing — DATABASE_URL host '${host}' is not a local database.\n` +
        `Allowed hosts: ${LOCAL_HOSTS.join(", ")}\n` +
        `Never set LOCAL_DEMO=1 with a remote database URL.`
    )
    process.exit(1)
  }
}

enforceLocalOnly()

const pool = new Pool({ connectionString: process.env.DATABASE_URL })
const adapter = new PrismaPg(pool)
const db = new PrismaClient({ adapter })

// ── Constants ────────────────────────────────────────────────────────────────
const ORG_ID = "org_axis_seed"
const DEMO_PASSWORD = "Demo1234!"
const DEMO_MARKER_SHA = "demo-seed-v1"

// ── Helper: password hash ────────────────────────────────────────────────────
async function hashPassword(raw: string): Promise<string> {
  return bcrypt.hash(raw, 12)
}

// ── Helper: UTC midnight of N days ago ───────────────────────────────────────
function daysAgo(n: number): Date {
  const d = new Date()
  d.setUTCHours(0, 0, 0, 0)
  d.setUTCDate(d.getUTCDate() - n)
  return d
}

// ── Helper: timestamp N days ago + H hours + M minutes ───────────────────────
function ts(daysBack: number, hoursBack: number, minutesBack = 0): Date {
  const now = new Date()
  now.setDate(now.getDate() - daysBack)
  now.setHours(now.getHours() - hoursBack)
  now.setMinutes(now.getMinutes() - minutesBack)
  return now
}

// ── Helper: SHA-256 for inputHash (demo placeholder, never from real inputs) ─
function demoHash(seed: string): string {
  return createHash("sha256").update(`demo:${ORG_ID}:${seed}`).digest("hex")
}

// ── Helper: compute claudeSpendUSD (Sonnet 4.6 pricing) ─────────────────────
function spend(input: number, output: number, cacheCreate: number, cacheRead: number): number {
  const raw = (input * 3.0 + output * 15.0 + cacheCreate * 3.75 + cacheRead * 0.3) / 1_000_000
  return Math.round(raw * 1_000_000) / 1_000_000
}

// ── Helper: upsert a User + Membership atomically ────────────────────────────
async function upsertDemoUser(
  email: string,
  name: string,
  role: "MANAGER" | "LINE_MANAGER" | "MEMBER",
  teamId: string | null,
  isLineManager: boolean
): Promise<string> {
  let user = await db.user.findUnique({ where: { email } })
  if (!user) {
    user = await db.user.create({
      data: { email, name, passwordHash: await hashPassword(DEMO_PASSWORD) },
    })
  }
  await db.membership.upsert({
    where: { userId_organisationId: { userId: user.id, organisationId: ORG_ID } },
    create: { userId: user.id, organisationId: ORG_ID, role, teamId, isLineManager, status: "ACTIVE" },
    update: { role, teamId, isLineManager, status: "ACTIVE" },
  })
  return user.id
}

async function seedUsers(): Promise<{
  managerId: string
  sarahId: string
  alexId: string
  priyaId: string
  benId: string
  danId: string
  engTeamId: string
}> {
  // Look up the Engineering team created by the base seed
  const engTeam = await db.team.findFirst({ where: { name: "Engineering", organisationId: ORG_ID } })
  if (!engTeam) throw new Error("Base seed not run — Engineering team not found. Run: npm run seed:0.5x first.")

  const engTeamId = engTeam.id

  // Enable time-tracking + AI time enrichment on Engineering team
  await db.team.update({
    where: { id: engTeamId },
    data: { timeTrackingEnabled: true, aiUsesTime: true },
  })

  // Existing MANAGER from base seed
  const managerUser = await db.user.findUnique({ where: { email: "manager@axis-pulse.dev" } })
  if (!managerUser) throw new Error("Base seed not run — manager@axis-pulse.dev not found.")
  const managerId = managerUser.id

  // Demo users
  const sarahId = await upsertDemoUser("sarah.kim@axis-pulse.dev",   "Sarah Kim",   "LINE_MANAGER", engTeamId, true)
  const alexId  = await upsertDemoUser("alex.torres@axis-pulse.dev", "Alex Torres", "MEMBER",       engTeamId, false)
  const priyaId = await upsertDemoUser("priya.mehta@axis-pulse.dev", "Priya Mehta", "MEMBER",       engTeamId, false)
  const benId   = await upsertDemoUser("ben.okafor@axis-pulse.dev",  "Ben Okafor",  "MEMBER",       engTeamId, false)
  const danId   = await upsertDemoUser("dan.walsh@axis-pulse.dev",   "Dan Walsh",   "MEMBER",       engTeamId, false)

  // Time-tracking consent (v2 = includes topApps) for all members
  for (const userId of [alexId, priyaId, benId, danId, sarahId]) {
    await db.timeTrackingConsent.upsert({
      where: { userId },
      create: { userId, acknowledgedAt: daysAgo(20), consentVersion: 2 },
      update: { acknowledgedAt: daysAgo(20), consentVersion: 2 },
    })
  }

  console.log("  Users seeded: sarah.kim, alex.torres, priya.mehta, ben.okafor, dan.walsh")
  return { managerId, sarahId, alexId, priyaId, benId, danId, engTeamId }
}
